// ------------------------------------------------------------
// HELPERS
// ------------------------------------------------------------
const toRoman = (n) => ["I", "II", "III", "IV", "V", "VI", "VII", "VIII", "IX", "X"][n - 1] || String(n);
const toLetter = (n) => String.fromCharCode(64 + n); // 1 -> A, 2 -> B...

let appendixRowIdCounter = 1;
const nextAppendixRowId = () => `apx-${appendixRowIdCounter++}`;

// Gán code_label cho từng node dựa trên vị trí trong cây — luôn tính
// lại từ mảng phẳng `items` (không lưu label vào state).
const computeLabels = (items) => {
  const labels = new Map();
  let h1 = 0;
  let itemCounter = 0;
  items.filter((i) => !i.parent_id).forEach((h1node) => {
    h1 += 1;
    labels.set(h1node.id, toRoman(h1));
    let h2 = 0;
    items.filter((i) => i.parent_id === h1node.id).forEach((h2node) => {
      h2 += 1;
      labels.set(h2node.id, toLetter(h2));
      items.filter((i) => i.parent_id === h2node.id).forEach((it) => {
        itemCounter += 1;
        labels.set(it.id, String(itemCounter));
      });
    });
  });
  return labels;
};

// Xóa 1 node và toàn bộ con cháu của nó (đệ quy)
const removeNodeAndChildren = (items, id) => {
  const idsToRemove = new Set([id]);
  let changed = true;
  while (changed) {
    changed = false;
    items.forEach((i) => {
      if (i.parent_id && idsToRemove.has(i.parent_id) && !idsToRemove.has(i.id)) {
        idsToRemove.add(i.id);
        changed = true;
      }
    });
  }
  return items.filter((i) => !idsToRemove.has(i.id));
};

// Tổng tiền toàn phụ lục — chỉ cộng dòng ITEM (HEADER không có tiền)
const quoteGrandTotal = (items) =>
  items.filter((i) => i.item_type === "ITEM").reduce((sum, i) => sum + (i.total_amount || 0), 0);

// ------------------------------------------------------------
// COMPONENT — AppendixTreeView (chỉ đọc, dùng để in/xem lại)
// ------------------------------------------------------------
const AppendixTreeView = ({ items }) => {
  const labels = computeLabels(items);
  const roots = items.filter((i) => !i.parent_id);

  const renderRow = (node, level) => (
    <React.Fragment key={node.id}>
      <tr>
        <td className="mono">{labels.get(node.id)}</td>
        <td style={{ paddingLeft: 12 + level * 20, fontWeight: node.item_type === "ITEM" ? 500 : 700 }}>
          {node.description}
          {node.item_type === "HEADER_1" && node.locations?.length > 0 && (
            <div style={{ fontSize: 11, fontWeight: 400, color: "var(--ink-faint)", marginTop: 2 }}>
              Địa điểm quan trắc: {node.locations.join("; ")}
            </div>
          )}
        </td>
        <td className="mono">{node.item_type === "ITEM" ? node.unit || "Mẫu" : ""}</td>
        <td className="mono">{node.item_type === "ITEM" ? node.sample_qty : ""}</td>
        <td className="mono">{node.item_type === "ITEM" ? node.batch_qty : ""}</td>
        <td className="mono">{node.item_type === "ITEM" ? node.unit_price?.toLocaleString("vi-VN") : ""}</td>
        <td className="mono" style={{ fontWeight: node.item_type === "ITEM" ? 600 : 400 }}>
          {node.item_type === "ITEM" ? node.total_amount?.toLocaleString("vi-VN") : ""}
        </td>
      </tr>
      {items.filter((i) => i.parent_id === node.id).map((child) => renderRow(child, level + 1))}
    </React.Fragment>
  );

  return (
    <table className="lims-table">
      <thead>
        <tr>
          <th style={{ width: 40 }}>TT</th>
          <th>Hạng mục công việc</th>
          <th>Đơn vị</th>
          <th>SL mẫu quan trắc/đợt</th>
          <th>Số đợt lấy mẫu</th>
          <th>Đơn giá (đ)</th>
          <th>Thành tiền (đ)</th>
        </tr>
      </thead>
      <tbody>
        {roots.map((r) => renderRow(r, 0))}
        {roots.length === 0 && (
          <tr><td colSpan={7} style={{ padding: 16, color: "var(--ink-faint)" }}>Chưa có hạng mục nào.</td></tr>
        )}
      </tbody>
      <tfoot>
        <tr>
          <td colSpan={6} style={{ textAlign: "right", fontWeight: 700, borderBottom: "none" }}>Tổng cộng</td>
          <td className="mono" style={{ fontWeight: 700, borderBottom: "none" }}>
            {quoteGrandTotal(items).toLocaleString("vi-VN")} đ
          </td>
        </tr>
      </tfoot>
    </table>
  );
};

// ------------------------------------------------------------
// COMPONENT — AppendixTreeBuilder (nhập liệu, dùng trong BaoGiaPage)
// ------------------------------------------------------------
const AppendixTreeBuilder = ({ items, setItems }) => {
  const labels = computeLabels(items);
  const [locationDraft, setLocationDraft] = useState({}); // { [header1Id]: "text đang gõ" }
  const [pickingIndicatorFor, setPickingIndicatorFor] = useState(null); // header2Id đang chọn chỉ tiêu

  const addHeader1 = () => {
    setItems((prev) => [
      ...prev,
      { id: nextAppendixRowId(), parent_id: null, item_type: "HEADER_1", description: "", locations: [] },
    ]);
  };

  const addHeader2 = (h1Id, matrixType) => {
    setItems((prev) => [
      ...prev,
      {
        id: nextAppendixRowId(), parent_id: h1Id, item_type: "HEADER_2",
        description: matrixType, matrix_type: matrixType,
      },
    ]);
  };

  const addItem = (h2Id, indicator) => {
    setItems((prev) => [
      ...prev,
      {
        id: nextAppendixRowId(), parent_id: h2Id, item_type: "ITEM",
        description: indicator.name, parameter_code: indicator.code,
        unit: indicator.unit === "-" ? "Mẫu" : indicator.unit,
        sample_qty: 1, batch_qty: 1, unit_price: indicator.price,
        total_amount: 1 * 1 * indicator.price,
      },
    ]);
    setPickingIndicatorFor(null);
  };

  const updateHeader1Desc = (id, val) =>
    setItems((prev) => prev.map((i) => (i.id === id ? { ...i, description: val } : i)));

  const addLocation = (h1Id) => {
    const text = (locationDraft[h1Id] || "").trim();
    if (!text) return;
    setItems((prev) => prev.map((i) => (i.id === h1Id ? { ...i, locations: [...(i.locations || []), text] } : i)));
    setLocationDraft((d) => ({ ...d, [h1Id]: "" }));
  };
  const removeLocation = (h1Id, idx) =>
    setItems((prev) => prev.map((i) => (i.id === h1Id ? { ...i, locations: i.locations.filter((_, k) => k !== idx) } : i)));

  const updateItemQty = (id, field, val) =>
    setItems((prev) =>
      prev.map((i) => {
        if (i.id !== id) return i;
        const next = { ...i, [field]: Math.max(1, parseInt(val) || 1) };
        next.total_amount = next.sample_qty * next.batch_qty * next.unit_price;
        return next;
      })
    );

  const removeNode = (id) => setItems((prev) => removeNodeAndChildren(prev, id));

  return (
    <div style={{ display: "flex", flexDirection: "column", gap: 12 }}>
      {items.filter((i) => !i.parent_id).map((h1) => (
        <div key={h1.id} style={{ border: "1px solid var(--line)", borderRadius: 8, overflow: "hidden" }}>
          {/* HEADER_1 */}
          <div style={{ padding: "10px 14px", background: "var(--surface-alt, #F8FAFC)", display: "flex", flexDirection: "column", gap: 8 }}>
            <div style={{ display: "flex", alignItems: "center", gap: 8 }}>
              <span className="badge" style={{ background: "var(--primary-soft)", color: "var(--primary-dark)", fontWeight: 700 }}>{labels.get(h1.id)}</span>
              <input
                className="lims-input" style={{ flex: 1, fontWeight: 700 }}
                placeholder="VD: QUAN TRẮC MÔI TRƯỜNG ĐỊNH KỲ"
                value={h1.description}
                onChange={(e) => updateHeader1Desc(h1.id, e.target.value)}
              />
              <button className="lims-btn-icon" onClick={() => removeNode(h1.id)} title="Xóa cả mục này"><Trash2 size={14} /></button>
            </div>

            {/* Địa điểm quan trắc — có thể nhiều */}
            <div style={{ display: "flex", flexWrap: "wrap", gap: 6, alignItems: "center" }}>
              <MapPin size={13} color="var(--ink-faint)" />
              {(h1.locations || []).map((loc, idx) => (
                <span key={idx} className="badge" style={{ background: "var(--blue-soft)", color: "var(--blue)" }}>
                  {loc}
                  <X size={11} style={{ cursor: "pointer" }} onClick={() => removeLocation(h1.id, idx)} />
                </span>
              ))}
              <input
                className="lims-input" style={{ width: 200, fontSize: 12, padding: "4px 8px" }}
                placeholder="Thêm địa điểm, Enter để lưu"
                value={locationDraft[h1.id] || ""}
                onChange={(e) => setLocationDraft((d) => ({ ...d, [h1.id]: e.target.value }))}
                onKeyDown={(e) => e.key === "Enter" && addLocation(h1.id)}
              />
            </div>

            <button className="lims-btn lims-btn-ghost" style={{ padding: "4px 10px", fontSize: 12, alignSelf: "flex-start" }} onClick={() => addHeader2(h1.id, SAMPLE_TYPES[0])}>
              <Plus size={13} /> Thêm loại chỉ tiêu (nền mẫu)
            </button>
          </div>

          {/* HEADER_2 + ITEM */}
          <div style={{ padding: "10px 14px", display: "flex", flexDirection: "column", gap: 10 }}>
            {items.filter((i) => i.parent_id === h1.id).map((h2) => (
              <div key={h2.id} style={{ border: "1px solid var(--line)", borderRadius: 8 }}>
                <div style={{ padding: "8px 12px", display: "flex", alignItems: "center", gap: 8, borderBottom: "1px solid var(--line)" }}>
                  <span className="badge" style={{ background: "var(--surface-alt)", color: "var(--ink-soft)", fontWeight: 700 }}>{labels.get(h2.id)}</span>
                  <select
                    className="lims-input" style={{ fontSize: 12.5 }}
                    value={h2.matrix_type}
                    onChange={(e) =>
                      setItems((prev) => prev.map((i) => (i.id === h2.id ? { ...i, matrix_type: e.target.value, description: e.target.value } : i)))
                    }
                  >
                    {SAMPLE_TYPES.map((t) => <option key={t}>{t}</option>)}
                  </select>
                  <div style={{ flex: 1 }} />
                  <button className="lims-btn lims-btn-ghost" style={{ padding: "4px 10px", fontSize: 12 }} onClick={() => setPickingIndicatorFor(h2.id)}>
                    <Plus size={13} /> Thêm chỉ tiêu
                  </button>
                  <button className="lims-btn-icon" onClick={() => removeNode(h2.id)} title="Xóa nhóm này"><Trash2 size={14} /></button>
                </div>

                {pickingIndicatorFor === h2.id && (
                  <div style={{ padding: 10, borderBottom: "1px solid var(--line)", display: "flex", flexWrap: "wrap", gap: 6 }}>
                    {INDICATORS.filter((ind) => ind.sampleTypes.includes(h2.matrix_type))
                      .filter((ind) => !items.some((i) => i.parent_id === h2.id && i.parameter_code === ind.code))
                      .map((ind) => (
                        <button key={ind.code} className="lims-btn lims-btn-ghost" style={{ fontSize: 12, padding: "5px 10px" }} onClick={() => addItem(h2.id, ind)}>
                          {ind.name} <span className="mono" style={{ color: "var(--ink-faint)" }}>({ind.price.toLocaleString("vi-VN")}đ)</span>
                        </button>
                      ))}
                    <button className="lims-btn-icon" onClick={() => setPickingIndicatorFor(null)}><X size={13} /></button>
                  </div>
                )}

                <div style={{ padding: "4px 12px" }}>
                  {items.filter((i) => i.parent_id === h2.id).map((it) => (
                    <div key={it.id} style={{ display: "flex", alignItems: "center", gap: 10, padding: "6px 0", borderBottom: "1px solid var(--line)", fontSize: 12.5 }}>
                      <span className="mono" style={{ width: 20, color: "var(--ink-faint)" }}>{labels.get(it.id)}</span>
                      <span style={{ flex: 1, fontWeight: 500 }}>{it.description}</span>
                      <span className="mono" style={{ width: 50 }}>{it.unit}</span>
                      <label style={{ display: "flex", alignItems: "center", gap: 4, color: "var(--ink-faint)" }}>
                        SL mẫu
                        <input type="number" min={1} className="lims-input" style={{ width: 55 }} value={it.sample_qty} onChange={(e) => updateItemQty(it.id, "sample_qty", e.target.value)} />
                      </label>
                      <label style={{ display: "flex", alignItems: "center", gap: 4, color: "var(--ink-faint)" }}>
                        Số đợt
                        <input type="number" min={1} className="lims-input" style={{ width: 55 }} value={it.batch_qty} onChange={(e) => updateItemQty(it.id, "batch_qty", e.target.value)} />
                      </label>
                      <span className="mono" style={{ width: 90, textAlign: "right" }}>{it.unit_price.toLocaleString("vi-VN")}đ</span>
                      <span className="mono" style={{ width: 100, textAlign: "right", fontWeight: 600 }}>{it.total_amount.toLocaleString("vi-VN")}đ</span>
                      <button className="lims-btn-icon" onClick={() => removeNode(it.id)}><X size={13} /></button>
                    </div>
                  ))}
                  {items.filter((i) => i.parent_id === h2.id).length === 0 && (
                    <div style={{ padding: "8px 0", fontSize: 12, color: "var(--ink-faint)" }}>Chưa có chỉ tiêu nào trong nhóm này.</div>
                  )}
                </div>
              </div>
            ))}
          </div>
        </div>
      ))}

      <button className="lims-btn lims-btn-primary" style={{ alignSelf: "flex-start" }} onClick={addHeader1}>
        <Plus size={14} /> Thêm Mục lớn (Phụ lục)
      </button>

      {items.length > 0 && (
        <div style={{ display: "flex", justifyContent: "flex-end", fontSize: 14, fontWeight: 700, paddingTop: 4 }}>
          Tổng thành tiền:&nbsp;
          <span className="mono" style={{ color: "var(--primary-dark)", fontSize: 16 }}>
            {quoteGrandTotal(items).toLocaleString("vi-VN")} đ
          </span>
        </div>
      )}
    </div>
  );
};

  const [listSearch, setListSearch] = useState("");
  const [showAdvanced, setShowAdvanced] = useState(false);
  const [statusFilter, setStatusFilter] = useState("");

  const canEdit = role === "B" || role === "C";
  const hasValidItems = appendixItems.some((i) => i.item_type === "ITEM");

  const createQuote = () => {
    if (!hasValidItems) return;
    const code = `BG-${String(91 + quotes.length - 3).padStart(4, "0")}`;
    setQuotes((q) => [
      { code, kh: customer, ngay: new Date().toLocaleDateString("vi-VN"), freq, items: appendixItems, status: "Nháp" },
      ...q,
    ]);
    setAppendixItems([]);
  };

  const convertToOrder = (code) => {
    setQuotes((q) => q.map((x) => (x.code === code ? { ...x, status: "Đã chuyển đơn hàng" } : x)));
    setPage("TaoPhieuYCKN");
  };

  const filteredQuotes = quotes.filter(
    (q) => (q.code + q.kh).toLowerCase().includes(listSearch.toLowerCase()) && (!statusFilter || q.status === statusFilter)
  );

  return (
    <>
      <PageHeader title="Báo giá" subtitle="Lập báo giá dạng Phụ lục phân cấp: Mục lớn → Loại chỉ tiêu → Chỉ tiêu cụ thể" />
      <PageShell>
        <SectionCard title="Lập báo giá mới" icon={FileText}>
          <div style={{ display: "flex", flexWrap: "wrap", gap: 16, alignItems: "flex-end", marginBottom: 20 }}>
            <div>
              <label style={{ fontSize: 12, fontWeight: 600, color: "var(--ink-soft)", display: "block", marginBottom: 4 }}>Khách hàng</label>
              <SearchableSelect value={customer} onChange={(c) => setCustomer(c.name)} options={CUSTOMERS} getLabel={(c) => `${c.name} — ${c.id}`} placeholder="Gõ để tìm khách hàng..." />
            </div>
            <div>
              <label style={{ fontSize: 12, fontWeight: 600, color: "var(--ink-soft)", display: "block", marginBottom: 4 }}>Tần suất</label>
              <select className="lims-input" value={freq} onChange={(e) => setFreq(e.target.value)}>
                {["Hàng tháng (12 lần/năm)", "Hàng quý (4 lần/năm)", "3 tháng / lần (4 lần/năm)", "6 tháng / lần (2 lần/năm)", "Hàng năm (1 lần/năm)", "Theo yêu cầu"].map((f) => (
                  <option key={f}>{f}</option>
                ))}
              </select>
            </div>
          </div>

          <label style={{ fontSize: 13, fontWeight: 700, color: "var(--ink-dark)", display: "block", marginBottom: 10 }}>
            Phụ lục báo giá:
          </label>
          <AppendixTreeBuilder items={appendixItems} setItems={setAppendixItems} />

          <div style={{ display: "flex", gap: 8, marginTop: 16 }}>
            <button className="lims-btn lims-btn-ghost"><Printer size={14} /> In báo giá PDF</button>
            <button className="lims-btn lims-btn-primary" disabled={!canEdit || !hasValidItems} onClick={createQuote}>
              <Plus size={14} /> Lưu báo giá
            </button>
          </div>
        </SectionCard>

        <SectionCard title="Danh sách báo giá" icon={ClipboardList} style={{ padding: 0 }}>
          <div style={{ display: "flex", flexWrap: "wrap", gap: 10, alignItems: "center", padding: "14px 16px", borderBottom: "1px solid var(--line)" }}>
            <div style={{ position: "relative", flex: "1 1 220px", maxWidth: 320 }}>
              <Search size={14} color="var(--ink-faint)" style={{ position: "absolute", left: 10, top: 9 }} />
              <input className="lims-input" style={{ width: "100%", paddingLeft: 30 }} placeholder="Tìm theo mã báo giá, khách hàng..." value={listSearch} onChange={(e) => setListSearch(e.target.value)} />
            </div>
            <button className="lims-btn lims-btn-ghost" onClick={() => setShowAdvanced((s) => !s)}><ListFilter size={14} /> {showAdvanced ? "Ẩn bộ lọc" : "Bộ lọc nâng cao"}</button>
            <div style={{ flex: 1 }} />
            <button className="lims-btn lims-btn-ghost"><Download size={14} /> Xuất Excel</button>
          </div>
          {showAdvanced && (
            <div style={{ display: "flex", gap: 10, padding: "0 16px 14px" }}>
              <select className="lims-input" value={statusFilter} onChange={(e) => setStatusFilter(e.target.value)}>
                <option value="">Tất cả trạng thái</option>
                <option>Nháp</option><option>Đã gửi khách hàng</option><option>Đã chuyển đơn hàng</option>
              </select>
            </div>
          )}
          <div className="lims-scroll" style={{ overflowX: "auto" }}>
            <table className="lims-table">
              <thead>
                <tr><th>Mã báo giá</th><th>Khách hàng</th><th>Ngày lập</th><th>Tần suất</th><th>Số chỉ tiêu</th><th>Trạng thái</th><th></th></tr>
              </thead>
              <tbody>
                {filteredQuotes.map((q) => (
                  <tr key={q.code} style={{ cursor: q.status === "Đã chuyển đơn hàng" ? "pointer" : "default" }} onClick={() => q.status === "Đã chuyển đơn hàng" && setPage("TaoPhieuYCKN")}>
                    <td><SpecimenTag>{q.code}</SpecimenTag></td>
                    <td style={{ fontWeight: 600 }}>{q.kh}</td>
                    <td className="mono" style={{ color: "var(--ink-soft)" }}>{q.ngay}</td>
                    <td style={{ fontSize: 12, color: "var(--ink-soft)" }}>{q.freq}</td>
                    <td>{q.items.filter((i) => i.item_type === "ITEM").length}</td>
                    <td>
                      <span className="badge" style={{
                        background: q.status === "Đã chuyển đơn hàng" ? "var(--primary-soft)" : q.status === "Đã gửi khách hàng" ? "var(--blue-soft)" : "var(--gray-soft)",
                        color: q.status === "Đã chuyển đơn hàng" ? "var(--primary-dark)" : q.status === "Đã gửi khách hàng" ? "var(--blue)" : "#5B6659",
                      }}>{q.status}</span>
                    </td>
                    <td>
                      <div style={{ display: "flex", gap: 6 }}>
                        <button className="lims-btn lims-btn-ghost" style={{ padding: "5px 10px" }} onClick={(e) => { e.stopPropagation(); setExportQuote(q); }}>
                          <Download size={13} /> Xuất Excel
                        </button>
                        {q.status !== "Đã chuyển đơn hàng" && canEdit && (
                          <button className="lims-btn lims-btn-primary" style={{ padding: "5px 10px" }} onClick={(e) => { e.stopPropagation(); convertToOrder(q.code); }}>
                            <ArrowRightLeft size={13} /> Chuyển đơn hàng
                          </button>
                        )}
                      </div>
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        </SectionCard>
      </PageShell>

      {exportQuote && <QuotePrintModal quote={exportQuote} onClose={() => setExportQuote(null)} />}
    </>
  );
};
